# TgVault backend dependencies
#
# Два правила, по которым составлен этот файл:
#
# 1. Здесь только то, что реально импортируется кодом. Раньше список содержал
#    Jinja2, Pillow, rich, orjson, httpx, aiofiles и python-multipart — ни один
#    из них не используется (просмотрщик собирается строковыми шаблонами,
#    сериализация — stdlib json, файлы пишет сам Telethon). Лишние пакеты не
#    просто занимают место: каждый пакет с C/Rust-расширением — это ещё один
#    шанс, что установка упадёт на свежем Python, где колёс ещё нет.
#
# 2. Версии заданы диапазонами, а не точными пинами. Нижняя граница — то, на
#    чём всё проверено; верхняя — следующий мажор. Так проект ставится и на
#    Python 3.12, и на 3.14 (Fedora 43), где для старых точных версий колёс
#    может не быть, а pip не имеет права уйти на несовместимый мажор.
#
# Проверить разрешимость, ничего не устанавливая:
#     pip install --dry-run -r requirements.txt

# --- Web framework / ASGI server ---------------------------------------------
# Намеренно plain uvicorn, а не uvicorn[standard]: extra тянет httptools,
# uvloop и watchfiles — три C-расширения, которые чаще всего и не собираются на
# только что вышедшем Python. Для локального десктопного приложения выигрыш в
# скорости от них незначим, а риск установки — вполне реальный.
# websockets нужен явно: без него uvicorn не поднимет WebSocket-эндпоинт /ws.
fastapi>=0.115,<1.0
uvicorn>=0.30,<1.0
# Верхняя граница не косметическая: в websockets 17 убран legacy-слой,
# на который опирается ws-реализация uvicorn.
websockets>=14,<16

# --- Конфигурация и валидация ------------------------------------------------
# pydantic v2 — DTO из §4 контракта; pydantic-settings читает .env (§7).
pydantic>=2.9,<3
pydantic-settings>=2.5,<3
python-dotenv>=1.0,<2

# --- База данных (MySQL 8, async) --------------------------------------------
# SQLAlchemy 2.0 в async-режиме поверх aiomysql; PyMySQL нужен aiomysql как
# протокольный слой.
SQLAlchemy[asyncio]>=2.0.30,<2.1
aiomysql>=0.2.0,<1
PyMySQL>=1.1,<2

# --- Шифрование --------------------------------------------------------------
# Fernet из cryptography шифрует StringSession Telegram в accounts.session_enc.
# Без верхней границы: API Fernet стабилен, а свежие мажоры как раз и приносят
# колёса под новые версии Python.
cryptography>=42

# --- Кэш / шина событий ------------------------------------------------------
# redis-py (asyncio) для pub/sub прогресса задач; при недоступности — in-memory
# fallback. Нужна версия с .aclose() (>=5.0.1).
redis>=5.0.1,<7

# --- Telegram (MTProto) ------------------------------------------------------
# Telethon — user-account клиент: вход по коду/QR/2FA, iter_messages,
# download_media. Чистый Python, ставится везде.
Telethon>=1.41,<2
